1. Write a JavaScript program to reverse a given number without converting it into a string.
	input : let num = 12345;
	Output: 54321

Answer: 
let num = 12345;
let reverse = 0;

while(num>0){

let digit = num % 10;

reverse = reverse * 10 + digit ;

num = Math.floor(num / 10);

}

console.log(reverse);

Explanation:
num = 12345

digit = 5
reverse = 0 * 10 + 5
reverse = 5

digit = 4
reverse = 5 * 10 + 4
reverse = 54

digit = 3
reverse = 54 * 10 + 3
reverse = 543

digit = 2
reverse = 543 * 10 + 2
reverse = 5432

digit = 1
reverse = 5432 * 10 + 1
reverse = 54321

2. Write a program to check whether a number is a palindrome.
Input : 121
Output: 121

Program: 
let num = 121;
let original = num;
let reverse = 0;

while(num>0){

let digit = num % 10;

reverse = reverse * 10 + digit ;

num = Math.floor(num / 10);

}

if(original === reverse)
console.log("Palindrome");
else
console.log("not Palindrome");

        let reverse = 0;
        let original = num;

            reverse = reverse * 10 + (num % 10);
            num = Math.floor(num / 10);
        }
        (original === reverse) ? console.log("Palindrome") : console.log("not Palindrome");

input :

output : 
Largest : 78

let b = 78;
let c = 32;

if(a >= b && a>=c)

	largest = a;

4. Write a program to check whether a number is prime.

Input:
let num = 29;
Output :
Prime


let num = 29;
let isPrime = true;

if(num <= 1)
{
isPrime = false;
}
else{

	for(let i = 2; i < num ; i++)
	{
		if(num % i === 0 )
		{
			isPrime = false;
			break;
		}
	}
}

(isPrime) ? console.log("Prime") : console.log("Not Prime");
