1. Write a JavaScript program to reverse a given number without converting it into a string.
	input : let num = 12345;
	Output: 54321

Answer: 
let num = 12345;
let reverse = 0;

while(num>0){

let digit = num % 10;

reverse = reverse * 10 + digit ;

num = Math.floor(num / 10);

}

console.log(reverse);

Explanation:
num = 12345

digit = 5
reverse = 0 * 10 + 5
reverse = 5

digit = 4
reverse = 5 * 10 + 4
reverse = 54

digit = 3
reverse = 54 * 10 + 3
reverse = 543

digit = 2
reverse = 543 * 10 + 2
reverse = 5432

digit = 1
reverse = 5432 * 10 + 1
reverse = 54321

2. Write a program to check whether a number is a palindrome.
Input : 121
Output: 121

Program: 
let num = 121;
let original = num;
let reverse = 0;

while(num>0){

let digit = num % 10;

reverse = reverse * 10 + digit ;

num = Math.floor(num / 10);

}

if(original === reverse)
console.log("Palindrome");
else
console.log("not Palindrome");

4. Write a program to check whether a number is prime.

Input:
let num = 29;
Output :
Prime


let num = 29;
let isPrime = true;

if(num <= 1)
{
isPrime = false;
}
else{

	for(let i = 2; i < num ; i++)
	{
		if(num % i === 0 )
		{
			isPrime = false;
			break;
		}
	}
}

(isPrime) ? console.log("Prime") : console.log("Not Prime");

7. Check whether a three-digit number is an Armstrong number.
Input : 153
		1³ + 5³ + 3³

		= 1 + 125 + 27
		= 153
Output: ArmstrongNumber


let num = 153;
        let sum = 0;
        let originalNum = num;


        while (num > 0) {
            let digit = num % 10;
            // sum = sum + digit * digit * digit;
            sum = sum + digit ** 3;
            num = Math.floor(num / 10);
        }
        if (sum === originalNum) {
            console.log(originalNum + " is an Armstrong number.");
        } else {
            console.log(originalNum + " is not an Armstrong number.");
        }

9. Count vowels and consonants in a string.

input : JavaScript
Output: vowels = 3
consonants = 7

let str = "Java Script is a programming language";
        let vowels = 0;
        let consonants = 0;

        for (let i = 0; i < str.length; i++) {

            let ch = str[i].toLowerCase();

            if (ch === " ")
                continue;

            if (ch === 'a' || ch === 'e' || ch === 'i' || ch == 'o' || ch == 'u') {
                vowels++;
            } else {
                consonants++;
            }
        }
        console.log("Vowels: " + vowels);
        console.log("Consonants: " + consonants);
